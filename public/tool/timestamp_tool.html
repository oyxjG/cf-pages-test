JavaScript / TypeScript 
// 获取秒级与毫秒级
const s = Math.floor(Date.now() / 1000);

Python 3 
import time, datetime
# 获取秒级时间戳
ts = int(time.time())
# 时间戳转 datetime
dt = datetime.datetime.fromtimestamp(ts)
Go (Golang) 
import "time"
// 获取秒级与纳秒
sec := time.Now().Unix()
msec := time.Now().UnixMilli()
// 转换时间
t := time.Unix(sec, 0)
Java 
// 获取毫秒级与秒级
long ms = System.currentTimeMillis();
